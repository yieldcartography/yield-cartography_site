% 2 ---------------------------------------------------------------------------
\begin{frame}{Motivation}
  \kicker{A point estimate conceals the uncertainty that matters}
  \begin{itemize}
    \item The discounted-cash-flow method delivers a single number,
          \( V = \sum_{t} \mathbb{E}[CF_t] \,/\, (1+r)^{t} \), from inputs
          that are themselves uncertain: projected cash flows and the
          discount rate.
    \item Sensitivity tables and scenario grids, the standard practice, probe
          the input space pointwise and carry no probability content.
    \item The paper reformulates the valuation to deliver the full
          distribution of outcomes implied by explicit distributions on the
          inputs.
  \end{itemize}
\end{frame}

% 3 ---------------------------------------------------------------------------
\begin{frame}{Density valuation}
  \kicker{Stochastic DCF, Monte Carlo and kernel density estimation}
  \small Cash flows are represented by revenue, cost, scaled and fixed
  components,
  \[ CF(t) \;=\; \sum_{i=1}^{N} R_i(t) - \sum_{i=1}^{M} C_i(t)
     + \sum_{i=1}^{K} S_i\big((R_n),(C_m),(Y_e),t\big) + F, \]
  with explicit stochastic processes on the drivers and on the discount rate,
  so the valuation $V$ becomes a random variable.
  \begin{itemize}
    \item The valuation density is generated parsimoniously by Monte Carlo
          simulation and estimated by kernel density methods; a
          Cornish--Fisher expansion is implemented as the analytic
          alternative, capturing skewness and kurtosis beyond the Gaussian
          approximation.
  \end{itemize}
\end{frame}

% 4 ---------------------------------------------------------------------------
\begin{frame}{From density to individual risk assessment}
  \kicker{Downside measures for heterogeneous users}
  \begin{itemize}
    \item The paper selects and discusses density-based downside risk
          measures that appraisers and analysts can attach to a DCF
          valuation: quantile-based bounds on value at chosen confidence
          levels and probabilities of falling short of critical levels.
    \item The target audience is a heterogeneous (by risk appetite) group of
          final users: each reads the same valuation density through their
          own risk tolerance, without relying on a single aggregated point
          recommendation.
    \item Valuation and risk measurement collapse into one object: the same
          density answers the pricing and the risk question.
  \end{itemize}
\end{frame}

% 5 ---------------------------------------------------------------------------
\begin{frame}{Implications}
  \kicker{For valuation practice}
  \begin{itemize}
    \item Reporting a valuation as a distribution disciplines the input
          assumptions: every scenario grid becomes a special case of an
          explicit probabilistic statement.
    \item Disagreements between valuations can be decomposed into
          disagreements about input distributions, which are auditable.
    \item The framework extends naturally to any present-value object with
          uncertain flows and rates, from equity valuation to project
          finance.
  \end{itemize}
\end{frame}
