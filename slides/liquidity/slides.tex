\begin{frame}{What actually moves bond liquidity?}
  \kicker{The question}
  \begin{itemize}
    \item In a less-liquid sovereign market, liquidity is not a fixed property
          of a bond. It is the joint output of issue size, supply, holder
          structure, the trading-venue rulebook and the macro regime.
    \item This paper treats liquidity as the \alert{dependent variable},
          not as an input to curve fitting.
    \item Five hypotheses, four data layers, twenty-one years of public data
          on the Polish sovereign curve.
  \end{itemize}
\end{frame}

% 3 ---------------------------------------------------------------------------
\begin{frame}{The setting: less-liquid markets (LLM)}
  \kicker{The setting}
  \begin{itemize}
    \item The middle tier of sovereign curves: visible enough that every dealer
          prices liquidity into a quote, transient enough that tick-data
          estimators built for the deepest curves do not transfer.
    \item Daily fixings and monthly turnover are the highest-frequency public
          record.
    \item Poland is the best-instrumented member of the cohort. Comparable
          LLMs: Brazil, Chile, Czechia, Hungary, India, Indonesia, Israel,
          Korea, Malaysia, Mexico, Singapore, South Africa, Thailand,
          T\"urkiye.
  \end{itemize}
\end{frame}

% 4 ---------------------------------------------------------------------------
\begin{frame}{Five hypotheses}
  \kicker{What we test}
  \small
  \begin{itemize}
    \item \textbf{H1. Issue size narrows everything.} Larger outstanding
          $\Rightarrow$ tighter spreads, lower price impact.
    \item \textbf{H2. Stocks, not flow.} Residual-maturity composition prices
          into liquidity, monthly issuance flow does not.
    \item \textbf{H3. Foreign retreat hits Amihud, not the quote.} Buy-side
          foreign holders exit; volume-scaled measures move, the venue-capped
          quote does not.
    \item \textbf{H4. MPC transmission is regime-dependent.} Pre- and
          post-COVID monetary days price into liquidity through different
          channels.
    \item \textbf{H5. Auctions leave an inventory imprint.} Primary Dealers
          absorb at concession and unwind into the secondary fixing.
  \end{itemize}
\end{frame}

% 5 ---------------------------------------------------------------------------
\begin{frame}{Finding 1: issue size is the most robust bond-level driver}
  \kicker{H1 supported}
  \bignum{$-3.5$ bp}{bond-month coefficient on log(outstanding) for the quoted
  spread, $p<0.01$. Same sign on the composite $z$-score and on the
  zero-trading-days share.}
  \vspace{1mm}
  \begin{center}\small\color{ycmuted}
  Survives bond fixed effects, macro and regime controls,\\
  across all six liquidity estimators.
  \end{center}
\end{frame}

% 6 ---------------------------------------------------------------------------
\begin{frame}{Finding 2: supply matters through stocks, not flow}
  \kicker{H2 supported}
  \begin{itemize}
    \item Monthly gross issuance does not move segment liquidity once stock
          measures are in.
    \item Trailing-twelve-month rolling issuance and the log of segment stock
          load monotonically; the residual-maturity share loads positively on
          Amihud and on zero-trading days.
  \end{itemize}
  \vspace{2mm}
  \begin{beamercolorbox}[rounded=true,sep=3mm]{block body}
    \centering\bfseries A well-supplied segment with an even maturity ladder
    trades tighter than a thin segment with a clumpy ladder.
  \end{beamercolorbox}
\end{frame}

\begin{frame}{Finding 5: the supply wedge is macro-relevant}
  \kicker{Aggregate price tag}
  \small Reprice the Polish fixed-cash-flow universe under the current curve,
  the BRW risk-neutral curve, and the curve net of the supply-attributable
  term-premium wedge from the 2019--2025 debt rise:
  \bignum{2.4\%}{of the present value of the 951 bn PLN fixed-cash-flow
  universe is the mark-to-market difference attributable to the supply-driven
  term premium accumulated over the 2019--2025 debt expansion. About 23 bn PLN.}
\end{frame}

\begin{frame}{How to read the result}
  \kicker{What it means}
  \small
  \begin{itemize}
    \item \textbf{For practitioners.} Quote-based and volume-scaled liquidity
          metrics are not interchangeable when the venue caps the quote.
          Carry both: the decomposition is the information.
    \item \textbf{For policymakers.} The supply gradient on the term premium is
          real: a 100 bn PLN debt rise prices into the ten-year term premium by
          about 23 bp through the level channel, plus another 8 bp per
          percentage point of debt-to-GDP.
    \item \textbf{For researchers.} Treat sovereign liquidity on a less-liquid
          market as an endogenous object driven by stocks, holders and venue
          rules, not as a fixed bond attribute.
  \end{itemize}
\end{frame}

% 12 --------------------------------------------------------------------------
\begin{frame}{Methodology in one slide}
  \kicker{One panel, four layers}
  \small
  \begin{itemize}
    \item \textbf{Data.} Daily BondSpot fixings and turnover on 91
          PLN-denominated sovereign bonds, 2005--2026, plus Ministry of Finance
          holdings and a macro panel.
    \item \textbf{Measures.} Six low-frequency estimators: quoted bid-ask,
          Amihud, Roll, Bao--Pan--Wang gamma, Corwin--Schultz, composite
          $z$-score, plus the zero-trading-days share.
    \item \textbf{Econometrics.} Bond-month and segment-month panels with fixed
          effects and Driscoll--Kraay errors; MPC event study with a
          pre/post-COVID split; auction event study on the yield path;
          aggregate monthly regressions of BAS and ACM term premia on supply.
    \item \textbf{Valuation.} A three-curve present-value scenario pricing the
          supply-attributable wedge on the fixed-cash-flow universe.
  \end{itemize}
\end{frame}
